cmake_minimum_required(VERSION 3.16)
project(adc_test_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(PROJECT_ROOT ${CMAKE_SOURCE_DIR}/../..)
set(DEV_ADC_PORT "mock" CACHE STRING "" FORCE)
add_subdirectory(${PROJECT_ROOT}/drivers/dev_common ${CMAKE_BINARY_DIR}/dev_common)
add_subdirectory(${PROJECT_ROOT}/drivers/dev_adc ${CMAKE_BINARY_DIR}/dev_adc)
add_executable(adc_test_host test_adc.c)
target_include_directories(adc_test_host PRIVATE ${PROJECT_ROOT}/drivers/dev_adc/port/mock)
target_link_libraries(adc_test_host PRIVATE dev_common dev_adc)
target_compile_options(adc_test_host PRIVATE -Wall -Wextra -Werror -pedantic)
