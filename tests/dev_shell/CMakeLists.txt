cmake_minimum_required(VERSION 3.16)
project(shell_test_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(PROJECT_ROOT ${CMAKE_SOURCE_DIR}/../..)
set(DEV_UART_PORT "mock" CACHE STRING "" FORCE)
add_subdirectory(${PROJECT_ROOT}/drivers/dev_common ${CMAKE_BINARY_DIR}/dev_common)
add_subdirectory(${PROJECT_ROOT}/drivers/dev_ringbuf ${CMAKE_BINARY_DIR}/dev_ringbuf)
add_subdirectory(${PROJECT_ROOT}/drivers/dev_uart ${CMAKE_BINARY_DIR}/dev_uart)
add_subdirectory(${PROJECT_ROOT}/services/svc_shell ${CMAKE_BINARY_DIR}/svc_shell)
add_executable(shell_test_host test_shell.c)
target_include_directories(shell_test_host PRIVATE ${PROJECT_ROOT}/drivers/dev_uart/port/mock)
target_link_libraries(shell_test_host PRIVATE dev_common dev_ringbuf dev_uart svc_shell)
target_compile_options(shell_test_host PRIVATE -Wall -Wextra -Werror -pedantic)
